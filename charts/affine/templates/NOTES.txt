1. Get the application URL by running these commands:
{{- if .Values.app.externalUrl }}
  {{ .Values.app.externalUrl }}
{{- else if .Values.front.ingress.enabled }}
{{- range $host := .Values.front.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.front.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if .Values.front.httpRoute.enabled }}
{{- range $hostname := .Values.front.httpRoute.hostnames }}
  http://{{ $hostname }}
{{- end }}
{{- else if contains "NodePort" .Values.front.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "front.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.front.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "front.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "front.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.front.service.port }}
{{- else if contains "ClusterIP" .Values.front.service.type }}
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "front.fullname" . }} 8080:{{ .Values.front.service.port }}
{{- end }}
